/* =====================================================
   CSS POSITIONING TECHNIQUES
   1. static   2. relative   3. absolute   4. fixed   5. sticky
   ===================================================== */

/* 1. STATIC – the default for every element.
   Elements follow the normal document flow; top/left/z-index have no effect. */
.hero-text,
.section-intro,
.info-list {
  position: static;
}

/* 2. RELATIVE – the element stays in the flow but can be nudged
   from its normal place. It also becomes the reference box for
   absolutely positioned children (badge, accent line). */

  top: -6px;                /* lifted 6px; the space it originally took is kept */
